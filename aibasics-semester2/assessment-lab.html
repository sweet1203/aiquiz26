import pandas as pd
import matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import confusion_matrix, classification_report
iris = load_iris(as_frame=True)
df = iris.frame.copy()
df.columns = ['sepal_len', 'sepal_wid', 'petal_len', 'petal_wid', 'target']
df['is_virginica'] = (df['target'] == 2).astype(int)
df = df.drop(columns=['target'])
print(df.head())
target = 'is_virginica'
수치열 = ['sepal_len', 'sepal_wid', 'petal_len', 'petal_wid']
print(df.isnull().sum())
df = df.dropna()
df['sepal_len'] = df['sepal_len'].fillna(df['sepal_len'].median())
df['sepal_len'] = df['sepal_len'].fillna(df['sepal_len'].mean())
df['sepal_len'] = df['sepal_len'].fillna(df['sepal_len'].mode()[0])
df[수치열].plot(kind='box')
plt.title('outliers')
plt.show()
corr = df[수치열 + [target]].corr()
plt.imshow(corr, cmap='coolwarm', vmin=-1, vmax=1)
plt.colorbar()
plt.xticks(range(len(corr)), corr.columns, rotation=45, ha='right')
plt.yticks(range(len(corr)), corr.columns)
for i in range(len(corr)):
    for j in range(len(corr)):
        plt.text(j, i, round(corr.iloc[i, j], 2), ha='center', va='center')
plt.title('correlation')
plt.show()
print(corr[target])
핵심변수 = 'petal_len'
X = df[수치열]
y = df[target]
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=7, stratify=y)
model = DecisionTreeClassifier(random_state=7, max_depth=4)
model.fit(X_train, y_train)
print(round(model.score(X_test, y_test), 2))
pred = model.predict(X_test)
cm = confusion_matrix(y_test, pred, labels=[0, 1])
print(cm)
TN, FP = cm[0]
FN, TP = cm[1]
